# Ejercicio 2 — "La Caja de Números"

# Enunciado:

# Crea una clase llamada CajaNumeros que:

# Al crearse (__init__), empiece con una lista vacía por dentro.
# Tenga un método agregar(numero) que guarde ese número en la lista.
# Tenga un método total() que devuelva la suma de todos los números guardados.
class CajaNumeros:
    def __init__(self):
        self.lista = []
    def agregar(self, numero):
        self.lista.append(numero)
    def total(self):
        return sum(self.lista)
caja = CajaNumeros()
caja.agregar(5)
caja.agregar(10)
caja.agregar(3)
print(caja.total()) 
# 🎮 NIVEL 3 — "El Filtro de Edades"

# Enunciado:

# Crea una clase llamada FiltroEdades que:

# Al crearse, empiece con una lista vacía por dentro.
# Tenga un método agregar_edad(edad) que guarde esa edad en la lista.
# Tenga un método mayores_de_edad() que devuelva una nueva lista con solo las edades que sean 18 o más.
class FiltroEdades:
    def __init__(self):
        self.lista = []
    def agregar_edad(self, edad):
        self.lista.append(edad)
    def mayores_de_edad(self):
        self.nuevalista = []
        for i in self.lista:
            if i >= 18:
                self.nuevalista.append(i)

        return self.nuevalista

f = FiltroEdades()
f.agregar_edad(15)
f.agregar_edad(20)
f.agregar_edad(17)
f.agregar_edad(30)
print(f.mayores_de_edad()) 

# 🎮 NIVEL 4 — "El Traductor Básico"

# Enunciado:

# Crea una clase llamada Traductor que:

# Al crearse, empiece con un diccionario vacío por dentro (que guardará pares palabra en español → palabra en inglés).
# Tenga un método agregar_traduccion(palabra_es, palabra_en) que guarde ese par en el diccionario.
# Tenga un método traducir(palabra_es) que devuelva la traducción en inglés de esa palabra. 
# Si la palabra no existe en el diccionario, debe devolver el texto "No encontrado".
class Traductor:
    def __init__(self):
        self.diccionario = {}
    def agregar_traduccion(self, palabra_es, palabra_en):
        self.diccionario[palabra_es] = palabra_en
    def traducir(self, palabra_es):
        if palabra_es in self.diccionario:
            return self.diccionario[palabra_es]
        else:
            return "No encontrado"
t = Traductor()
t.agregar_traduccion("hola", "hello")
t.agregar_traduccion("gato", "cat")
print(t.traducir("hola"))    
print(t.traducir("perro"))

# 🎮 NIVEL 5 — "El Verificador de Contraseñas"

# Enunciado:

# Crea una clase llamada VerificadorContraseñas que:

# Tenga un método es_valida(contraseña) que devuelva True si la contraseña tiene 8 caracteres o más, y False si no.
# Tenga un método verificar_multiples(*contraseñas) que reciba varias contraseñas y devuelva un diccionario {contraseña: True/False}, 
# reutilizando es_valida para cada una (no reescribas la lógica de validar adentro de este método).
class VerificadorContraseñas:
    def es_valida(self, contraseña):
        if len(contraseña) >= 8:
            return True
        else:
            return False
    def verificar_multiples(self, *contraseña):
        mi_diccionario = {}
        for i in contraseña:
            mi_diccionario[i] = self.es_valida(i)

        return mi_diccionario
v = VerificadorContraseñas()
print(v.verificar_multiples("1234", "miclaveSegura123", "abc"))
